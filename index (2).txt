  <script>
    function log(m) {
      const b = document.getElementById('sys-log');
      if (b) b.innerHTML = '&gt; ' + m + '<br>' + b.innerHTML;
    }
    function saveDB() {
      if (window.STIKS && window.STIKS.db) {
        localStorage.setItem('stiks_prod_db', JSON.stringify(window.STIKS.db));
      }
    }
    function toggleVoice() {
      if (!window.STIKS) return;
      window.STIKS.voiceOn = !window.STIKS.voiceOn;
      const btn = document.getElementById('btn-voice-toggle');
      if (btn) btn.innerText = window.STIKS.voiceOn ? '🔊 Озвучка: ВКЛ' : '🔇 Озвучка: ВЫКЛ';
      if (!window.STIKS.voiceOn && 'speechSynthesis' in window) window.speechSynthesis.cancel();
    }
    function exportSave() {
      if (!window.STIKS) return;
      const cur = JSON.stringify(window.STIKS.db);
      const d = prompt('Скопируйте код сохранения или вставьте чужой:', cur);
      if (d && d !== cur) {
        try {
          window.STIKS.db = JSON.parse(d);
          saveDB();
          location.reload();
        } catch (e) {
          alert('Ошибка формата сохранения');
        }
      }
    }
    window.onload = function() {
      if (window.STIKS) {
        const s = localStorage.getItem('stiks_prod_db');
        if (s) {
          try { window.STIKS.db = JSON.parse(s); } catch (e) {}
        }
        if (window.STIKS_PATCHER && window.STIKS_PATCHER.boot) {
          const info = window.STIKS_PATCHER.boot();
          const badge = document.getElementById('patch-status');
          if (badge && info) badge.innerText = info;
        }
        if (window.STIKS.db.session && window.STIKS.renderGame) {
          window.STIKS.renderGame();
        } else if (window.STIKS.renderLobby) {
          window.STIKS.renderLobby();
        }
      } else {
        log('Ошибка загрузки модулей stiks_db.js / stiks_engine.js');
      }
    };
  </script>
