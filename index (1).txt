  <script>
    function log(m) {
      const b = document.getElementById('sys-log');
      if (b) b.innerHTML = '&gt; ' + m + '<br>' + b.innerHTML;
    }

    function hasNonEmptyFamilies(obj) {
      return !!(obj && obj.families && typeof obj.families === 'object' && Object.keys(obj.families).length > 0);
    }

    function saveDB() {
      if (!window.STIKS || !window.STIKS.db) return;
      const curDb = window.STIKS.db;
      const rawPrev = localStorage.getItem('stiks_prod_db');
      let prevDb = null;
      if (rawPrev) {
        try { prevDb = JSON.parse(rawPrev); } catch (e) {}
      }
      if (!hasNonEmptyFamilies(curDb) && hasNonEmptyFamilies(prevDb)) {
        window.STIKS.db.families = prevDb.families;
        return;
      }
      const serialized = JSON.stringify(window.STIKS.db);
      localStorage.setItem('stiks_prod_db', serialized);
      if (hasNonEmptyFamilies(window.STIKS.db)) {
        localStorage.setItem('stiks_prod_db_safe_backup', serialized);
      }
    }

    function toggleVoice() {
      if (!window.STIKS) return;
      window.STIKS.voiceOn = !window.STIKS.voiceOn;
      const btn = document.getElementById('btn-voice-toggle');
      if (btn) btn.innerText = window.STIKS.voiceOn ? '🔊 Озвучка: ВКЛ' : '🔇 Озвучка: ВЫКЛ';
      if (!window.STIKS.voiceOn && 'speechSynthesis' in window) window.speechSynthesis.cancel();
    }

    function downloadHeroesBackup() {
      if (!window.STIKS || !window.STIKS.db) return;
      saveDB();
      const raw = localStorage.getItem('stiks_prod_db') || JSON.stringify(window.STIKS.db);
      const blob = new Blob([raw], { type: 'application/json;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      const dt = new Date().toISOString().slice(0, 10);
      a.href = url;
      a.download = 'stiks_heroes_' + dt + '.json';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(function() { URL.revokeObjectURL(url); }, 1000);
      log('💾 Бэкап персонажей сохранён в файл (.json).');
    }

    function uploadHeroesBackup(input) {
      const file = input.files && input.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const parsed = JSON.parse(e.target.result);
          if (!parsed || typeof parsed !== 'object' || !parsed.families) {
            throw new Error('Неверная структура файла сохранения');
          }
          window.STIKS.db = parsed;
          const serialized = JSON.stringify(parsed);
          localStorage.setItem('stiks_prod_db', serialized);
          if (hasNonEmptyFamilies(parsed)) {
            localStorage.setItem('stiks_prod_db_safe_backup', serialized);
          }
          log('📂 Персонажи загружены из файла. Обновление...');
          setTimeout(function() { location.reload(); }, 250);
        } catch (err) {
          alert('Ошибка загрузки файла сохранения: ' + err.message);
        }
        input.value = '';
      };
      reader.readAsText(file, 'UTF-8');
    }

    window.onload = function() {
      if (window.STIKS) {
        let loaded = null;
        const s = localStorage.getItem('stiks_prod_db');
        const b = localStorage.getItem('stiks_prod_db_safe_backup');
        if (s) {
          try { loaded = JSON.parse(s); } catch (e) {}
        }
        if (!hasNonEmptyFamilies(loaded) && b) {
          try {
            const backupObj = JSON.parse(b);
            if (hasNonEmptyFamilies(backupObj)) {
              loaded = backupObj;
              localStorage.setItem('stiks_prod_db', JSON.stringify(loaded));
            }
          } catch (e) {}
        }
        if (loaded) {
          window.STIKS.db = loaded;
        }
        if (window.STIKS_PATCHER && window.STIKS_PATCHER.boot) {
          const info = window.STIKS_PATCHER.boot();
          const badge = document.getElementById('patch-status');
          if (badge && info) badge.innerText = info;
        }
        if (window.STIKS.db && window.STIKS.db.session && window.STIKS.renderGame) {
          window.STIKS.renderGame();
        } else if (window.STIKS.renderLobby) {
          window.STIKS.renderLobby();
        }
      } else {
        log('Ошибка загрузки модулей stiks_db.js / stiks_engine.js');
      }
    };
