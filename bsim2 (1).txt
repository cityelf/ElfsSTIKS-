window.BSIM_AUDIO = {
  ctx: null,
  bgmOn: false,
  bgmTimer: null,
  stepIdx: 0,

  ensureCtx() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (AC) this.ctx = new AC();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume().catch(() => {});
    }
    return this.ctx;
  },

  toggleBgm() {
    this.bgmOn = !this.bgmOn;
    const btn = document.getElementById('btn-bgm');
    if (btn) btn.textContent = this.bgmOn ? '🎵 Музыка: ВКЛ' : '🎵 Музыка: ВЫКЛ';
    if (this.bgmOn) {
      this.ensureCtx();
      this.stepIdx = 0;
      this.startLoop();
    } else if (this.bgmTimer) {
      clearInterval(this.bgmTimer);
      this.bgmTimer = null;
    }
  },

  startLoop() {
    if (this.bgmTimer) clearInterval(this.bgmTimer);
    // Цикл в полтона от SFX: "пас-дам-бас, парам-бам, парара-парамбам! пу-руру, пу-руру!"
    const seq = [
      { f: 110.0, d: 0.16, t: 'triangle', v: 0.045 }, // пас
      { f: 130.8, d: 0.16, t: 'triangle', v: 0.045 }, // дам
      { f: 98.0,  d: 0.24, t: 'triangle', v: 0.055 }, // бас
      { f: 0,     d: 0.10, t: 'sine',     v: 0 },
      { f: 146.8, d: 0.13, t: 'triangle', v: 0.045 }, // па-
      { f: 164.8, d: 0.13, t: 'triangle', v: 0.045 }, // рам
      { f: 130.8, d: 0.22, t: 'triangle', v: 0.055 }, // бам
      { f: 0,     d: 0.10, t: 'sine',     v: 0 },
      { f: 164.8, d: 0.10, t: 'triangle', v: 0.045 }, // па-
      { f: 174.6, d: 0.10, t: 'triangle', v: 0.045 }, // ра-
      { f: 196.0, d: 0.12, t: 'triangle', v: 0.045 }, // ра-
      { f: 164.8, d: 0.11, t: 'triangle', v: 0.045 }, // па-
      { f: 146.8, d: 0.11, t: 'triangle', v: 0.045 }, // рам-
      { f: 110.0, d: 0.24, t: 'triangle', v: 0.055 }, // бам!
      { f: 0,     d: 0.12, t: 'sine',     v: 0 },
      { f: 329.6, d: 0.09, t: 'sine',     v: 0.032 }, // пу-
      { f: 392.0, d: 0.09, t: 'sine',     v: 0.032 }, // ру-
      { f: 440.0, d: 0.16, t: 'sine',     v: 0.036 }, // ру
      { f: 0,     d: 0.08, t: 'sine',     v: 0 },
      { f: 293.7, d: 0.09, t: 'sine',     v: 0.032 }, // пу-
      { f: 349.2, d: 0.09, t: 'sine',     v: 0.032 }, // ру-
      { f: 392.0, d: 0.18, t: 'sine',     v: 0.036 }, // ру!
      { f: 0,     d: 0.16, t: 'sine',     v: 0 }
    ];
    this.bgmTimer = setInterval(() => {
      if (!this.bgmOn) return;
      const c = this.ensureCtx();
      if (!c) return;
      const note = seq[this.stepIdx % seq.length];
      this.stepIdx++;
      if (note.f > 0) {
        const now = c.currentTime;
        const o = c.createOscillator();
        const g = c.createGain();
        o.type = note.t;
        o.frequency.setValueAtTime(note.f, now);
        g.gain.setValueAtTime(note.v, now);
        g.gain.exponentialRampToValueAtTime(0.001, now + note.d);
        o.connect(g);
        g.connect(c.destination);
        o.start(now);
        o.stop(now + note.d + 0.01);
      }
    }, 210);
  },

  playSfx(kind) {
    const c = this.ensureCtx();
    if (!c) return;
    const now = c.currentTime;
    if (kind === 'gun') {
      // Выстрел ("Бах" / "Тра-та-та")
      const buf = c.createBuffer(1, c.sampleRate * 0.16, c.sampleRate);
      const data = buf.getChannelData(0);
      for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (c.sampleRate * 0.04));
      const src = c.createBufferSource();
      src.buffer = buf;
      const g = c.createGain();
      g.gain.setValueAtTime(0.18, now);
      g.gain.exponentialRampToValueAtTime(0.001, now + 0.16);
      src.connect(g);
      g.connect(c.destination);
      src.start(now);
    } else if (kind === 'melee') {
      // Взмах / удар ("Вжик" / "Дыщ")
      const o = c.createOscillator();
      const g = c.createGain();
      o.type = 'sawtooth';
      o.frequency.setValueAtTime(420, now);
      o.frequency.exponentialRampToValueAtTime(75, now + 0.14);
      g.gain.setValueAtTime(0.14, now);
      g.gain.exponentialRampToValueAtTime(0.001, now + 0.15);
      o.connect(g);
      g.connect(c.destination);
      o.start(now);
      o.stop(now + 0.16);
    } else if (kind === 'bite') {
      // Укус / царапок ("Цап" / "Кусь")
      const o = c.createOscillator();
      const g = c.createGain();
      o.type = 'square';
      o.frequency.setValueAtTime(160, now);
      o.frequency.exponentialRampToValueAtTime(52, now + 0.13);
      g.gain.setValueAtTime(0.13, now);
      g.gain.exponentialRampToValueAtTime(0.001, now + 0.14);
      o.connect(g);
      g.connect(c.destination);
      o.start(now);
      o.stop(now + 0.15);
    } else if (kind === 'crit' || kind === 'dar') {
      // Крит или Дар Улья
      [523.25, 783.99].forEach((freq, idx) => {
        const o = c.createOscillator();
        const g = c.createGain();
        o.type = 'triangle';
        o.frequency.setValueAtTime(freq, now + idx * 0.07);
        g.gain.setValueAtTime(0.16, now + idx * 0.07);
        g.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.07 + 0.18);
        o.connect(g);
        g.connect(c.destination);
        o.start(now + idx * 0.07);
        o.stop(now + idx * 0.07 + 0.19);
      });
    }
  }
};

window.STIKS_COMBAT_CORE = {
  state: null,
  calcMana(m1, m30, lvl) {
    if (!m1 && !m30) return 0;
    const ratio = (Math.max(1, Math.min(30, lvl)) - 1) / 29;
    return Math.max(0, Math.round(m1 - (m1 - m30) * ratio));
  },
  getMult(wCls, eRank) {
    const d = wCls - eRank;
    if (wCls === 2) return d > 0 ? 3 : (d === 0 ? 2 : (d === -1 ? 1.5 : 1));
    if (wCls === 3) return d > 0 ? 2 : (d === 0 ? 3 : 1);
    if (wCls >= 4 && d >= 0) return 3;
    return 1;
  },
  start(cfg) {
    const D = window.BSIM_DATA;
    const arch = D.ARCHETYPES[cfg.archKey] || D.ARCHETYPES.kachok;
    const weap = D.WEAPONS[cfg.weapKey] || D.WEAPONS.gun_2;
    let img = arch.img || 'img/kachok.jpeg';
    if (cfg.archKey === 'psih') img = cfg.sex === 'f' ? arch.imgF : arch.imgM;
    if (cfg.archKey === 'kvaz') img = cfg.sex === 'f' ? arch.imgF : arch.imgM;

    const lvl = Math.max(0, Number(cfg.lvl) || 0);
    const statAvg = Math.round(10 + lvl * 4.5);
    const maxMana = Math.max(100, 10 + statAvg * 2);
    const infHp = 1000000;
    const rList = (cfg.enemyRanks && cfg.enemyRanks.length) ? cfg.enemyRanks : new Array(2).fill(2);

    const enemies = rList.map((rNum, idx) => {
      const zInfo = D.Z_RANKS.find(z => z.r === rNum) || D.Z_RANKS.at(0);
      return {
        id: 'en_' + idx,
        name: zInfo.name + (rList.length > 1 ? ' #' + (idx + 1) : ''),
        rank: zInfo.r,
        hp: infHp,
        maxHp: infHp,
        agi: zInfo.agi,
        dmg: zInfo.dmg,
        sound: zInfo.sound,
        sfx: zInfo.sfx || 'bite',
        stun: 0
      };
    });

    const maxERank = enemies.reduce((m, e) => Math.max(m, e.rank), 0);
    const dist = weap.isGun ? Math.max(1, 2 + (weap.cls - maxERank)) : 0;

    this.state = {
      loc: cfg.loc,
      round: 1,
      dist: dist,
      maxDist: Math.max(1, dist),
      player: {
        name: arch.name,
        archKey: cfg.archKey,
        lvl: lvl,
        img: img,
        hp: infHp,
        maxHp: infHp,
        mana: maxMana,
        maxMana: maxMana,
        stat: statAvg,
        weap: weap,
        armCls: cfg.armCls,
        armFlat: D.ARMOR_FLAT[cfg.armCls] || 0,
        skill: arch.skill,
        skillType: arch.skillType || 'active',
        phrase: arch.phrase,
        learned: JSON.parse(JSON.stringify(cfg.learned || [])),
        buff: 1
      },
      enemies: enemies
    };
    return this.state;
  }
};

      dSel.value = 'ice_master';
    }
    this.syncBranches();
    this.renderSetupLists();
  },

  syncBranches() {
    const D = window.BSIM_DATA;
    const dKey = (document.getElementById('sim-dar') || {}).value || '';
    const bSel = document.getElementById('sim-branch');
    if (!bSel || !D.DARS[dKey]) return;
    bSel.innerHTML = '<option value="">[База] ' + D.DARS[dKey].name + '</option>' +
      D.DARS[dKey].branches.map(b => '<option value="' + b.id + '">[Ветка] ' + b.name + '</option>').join('');
  },

  addLearnedGift() {
    const dKey = (document.getElementById('sim-dar') || {}).value || '';
    const branchId = (document.getElementById('sim-branch') || {}).value || '';
    const lvl = Math.max(1, Math.min(30, Number((document.getElementById('sim-darlvl') || {}).value) || 1));
    if (!dKey) return;
    const exists = this.learned.find(x => x.darKey === dKey && x.branchId === branchId);
    if (exists) exists.lvl = lvl;
    else this.learned.push({ darKey: dKey, branchId: branchId, lvl: lvl });
    this.renderSetupLists();
  },

  removeLearned(idx) {
    this.learned.splice(idx, 1);
    this.renderSetupLists();
  },

  addEnemyToRoster() {
    if (this.enemyRanks.length >= 6) return;
    const r = Number((document.getElementById('sim-erank') || {}).value) || 0;
    this.enemyRanks.push(r);
    this.renderSetupLists();
  },

  removeEnemyRoster(idx) {
    if (this.enemyRanks.length <= 1) return;
    this.enemyRanks.splice(idx, 1);
    this.renderSetupLists();
  },

    const D = window.BSIM_DATA;
    const lBox = document.getElementById('learned-list');

          const dar = D.DARS[item.darKey];
          const br = item.branchId ? dar.branches.find(b => b.id === item.branchId) : null;
          const title = br ? (dar.name + ' → ' + br.name) : dar.name;
          const tp = br ? br.type : dar.type;
          const tag = tp === 'passive' ? 'ПАССИВ' : (tp === 'trigger' ? 'ТРИГГЕР' : 'АКТИВ');

    const unit = document.getElementById(unitDomId);
    const stage = document.getElementById('combat-stage');
    if (!layer || !unit || !stage) return;
    const uRect = unit.getBoundingClientRect();
    const sRect = stage.getBoundingClientRect();
    const el = document.createElement('div');
    el.className = 'float-pop ' + cls;
    el.textContent = txt;
    el.style.top = Math.max(22, Math.round(uRect.top - sRect.top - 4)) + 'px';
    if (isLeftSide) {
      el.style.left = Math.max(4, Math.round(uRect.left - sRect.left - 8)) + 'px';
    } else {
      el.style.right = Math.max(4, Math.round(sRect.right - uRect.right - 8)) + 'px';
    }
    layer.appendChild(el);
    setTimeout(() => { if (el.parentNode) el.remove(); }, 1250);
  },

  renderBattle() {
    const st = window.STIKS_COMBAT_CORE.state;
    if (!st) return;
    const D = window.BSIM_DATA;
    const p = st.player;
    const ratio = st.maxDist > 0 ? (st.dist / st.maxDist) : 0;
    const shiftPct = Math.round(4 + (1 - ratio) * 26);

    const colA = document.getElementById('col-allies');
    const colE = document.getElementById('col-enemies');
    if (colA) {
      colA.style.left = shiftPct + '%';
      const pFb = encodeURIComponent(D.getRankFallbackSvg(0, true));
      colA.innerHTML = '<div id="u-player" class="mini-unit ally"><div class="mini-pic"><img src="' + p.img + '" onerror="this.onerror=null;this.src=\'data:image/svg+xml;utf8,' + pFb + '\';"></div><div class="mini-meta"><div class="mini-lbl">' + p.name + '</div><div class="mini-bar"><div class="mini-fill ally" style="width:100%;"></div></div></div></div>';
    }
    if (colE) {
      colE.style.right = shiftPct + '%';
      colE.innerHTML = st.enemies.map(en => {
        const eFb = encodeURIComponent(D.getRankFallbackSvg(en.rank, false));
        return '<div id="u-' + en.id + '" class="mini-unit enemy' + (en.stun > 0 ? ' stun' : '') + '"><div class="mini-pic"><img src="data:image/svg+xml;utf8,' + eFb + '"></div><div class="mini-meta"><div class="mini-lbl">' + en.name + '</div><div class="mini-bar"><div class="mini-fill" style="width:100%;"></div></div></div></div>';
      }).join('');
    }

    const distEl = document.getElementById('stage-dist');
    if (distEl) distEl.textContent = st.dist <= 0 ? 'Раунд ' + st.round + ' • Вплотную' : 'Раунд ' + st.round + ' • Дистанция: ' + st.dist + ' кл.';

    const bars = document.getElementById('sim-bars');
    if (bars) {
      bars.innerHTML = '<div>❤️ ОЗ: <b>' + p.hp.toLocaleString('ru-RU') + '</b></div><div>🔮 Мана: <b>' + p.mana + '/' + p.maxMana + '</b></div><div>🛡 Броня: <b>' + p.armFlat + ' (Р' + p.armCls + ')</b></div>';
    }

    const abBox = document.getElementById('battle-abilities');
    if (abBox) {
      let abHtml = '';
      if (p.skillType === 'trigger' || p.skillType === 'passive') {
        abHtml += '<div id="badge-skill" class="passive-badge"><span>🛡 [Триггер] ' + p.skill + '</span><span>АВТО</span></div>';
      } else {
        abHtml += '<button class="btn" style="text-align:left;" onclick="BSIM_UI.actSkill()">✨ [Навык] ' + p.skill + '</button>';
      }
      p.learned.forEach((item, idx) => {
        const dar = D.DARS[item.darKey];
        if (!dar) return;
        const br = item.branchId ? dar.branches.find(b => b.id === item.branchId) : null;
        const title = br ? br.name : dar.name;
        const tp = br ? br.type : dar.type;
        const pct = window.STIKS_COMBAT_CORE.calcMana(br ? br.mana1 : dar.mana1, br ? br.mana30 : dar.mana30, item.lvl);
        const cost = Math.floor(p.maxMana * (pct / 100));
        if (tp === 'passive' || tp === 'trigger') {
          abHtml += '<div id="badge-dar-' + idx + '" class="passive-badge"><span>🧬 [' + (tp === 'trigger' ? 'Триггер' : 'Пассив') + '] ' + title + ' (Ур.' + item.lvl + ')</span><span>' + (cost > 0 ? cost + ' м.' : '0%') + '</span></div>';
        } else {
          abHtml += '<button class="btn btn-dar" onclick="BSIM_UI.actLearned(' + idx + ')">🧬 ' + title + ' (Ур.' + item.lvl + ' • ' + pct + '% / ' + cost + ' маны)</button>';
        }
      });
      abBox.innerHTML = abHtml;
    }
  },

  actAttack() {
    const st = window.STIKS_COMBAT_CORE.state;
    if (!st) return;
    const p = st.player;
    const trg = st.enemies.at(0);
    if (!trg) return;
    const d1 = Math.floor(Math.random() * 6) + 1;
    const d2 = Math.floor(Math.random() * 6) + 1;
    const crit = (d1 === d2) || (Math.random() < 0.2);
    const mult = window.STIKS_COMBAT_CORE.getMult(p.weap.cls, trg.rank);
    let dmg = Math.max(1, Math.round((p.weap.dmg + (d1 + d2) * 2 + p.stat * 0.4) * mult * p.buff));
    if (crit) dmg = Math.round(dmg * 1.75);
    trg.hp = Math.max(1, trg.hp - dmg);
    window.BSIM_AUDIO.playSfx(crit ? 'crit' : p.weap.sfx);
    this.popBack('u-player', true, p.weap.sound, 'float-sfx');
    this.popBack('u-' + trg.id, false, '-' + dmg + (crit ? '!' : ''), crit ? 'float-crit' : 'float-dmg');
    this.log(p.weap.sound + ' <b>' + p.name + '</b> [' + d1 + '+' + d2 + '] → <b>' + trg.name + '</b>: <b>-' + dmg + '</b>' + (crit ? ' <span style="color:#ff4444;">(КРИТ!)</span>' : ''));
    this.endRound();
  },

  actSkill() {
    const st = window.STIKS_COMBAT_CORE.state;
    if (!st) return;
    const p = st.player;
    const trg = st.enemies.at(0);
    if (!trg) return;
    const dmg = Math.round((p.weap.dmg * 1.6 + p.stat * 0.9) * p.buff);
    trg.hp = Math.max(1, trg.hp - dmg);
    trg.stun = 1;
    window.BSIM_AUDIO.playSfx('crit');
    this.popBack('u-player', true, '✨ ' + p.skill, 'float-skill');
    this.popBack('u-' + trg.id, false, '-' + dmg + '!', 'float-crit');
    this.log('✨ Навык <b>[' + p.skill + ']</b> («' + p.phrase + '») → <b>-' + dmg + '</b> + оглушение!');
    this.endRound();
  },

  actLearned(idx) {
    const st = window.STIKS_COMBAT_CORE.state;
    const D = window.BSIM_DATA;
    if (!st || !st.player.learned.at(idx)) return;
    const p = st.player;
    const item = p.learned.at(idx);
    const dar = D.DARS[item.darKey];
    const br = item.branchId ? dar.branches.find(b => b.id === item.branchId) : null;
    const title = br ? br.name : dar.name;
    const pct = window.STIKS_COMBAT_CORE.calcMana(br ? br.mana1 : dar.mana1, br ? br.mana30 : dar.mana30, item.lvl);
    const cost = Math.floor(p.maxMana * (pct / 100));
    if (p.mana < cost) {
      this.popBack('u-player', true, '❌ Мало маны!', 'float-crit');
      this.log('❌ Не хватает маны на <b>[' + title + ']</b> (нужно ' + cost + '). Нажмите «🔮 Мана 100%»!');
      return;
    }
    p.mana -= cost;
    window.BSIM_AUDIO.playSfx('dar');
    const dmg = Math.round((65 + p.stat * 1.1) * (1 + item.lvl * 0.08));
    st.enemies.forEach(en => {
      en.hp = Math.max(1, en.hp - dmg);
      this.popBack('u-' + en.id, false, '-' + dmg + '!', 'float-crit');
    });
    this.popBack('u-player', true, '🧬 ' + title, 'float-skill');
    this.log('🧬 Дар <b>[' + title + ']</b> (Ур.' + item.lvl + ', -' + cost + ' маны): урон по всем <b>-' + dmg + '</b>!');
    this.endRound();
  },

  endRound() {
    const st = window.STIKS_COMBAT_CORE.state;
    const p = st.player;
    if (st.dist > 0) {
      st.dist = Math.max(0, st.dist - 1);
      this.log('🏃 Враги сближаются → дистанция <b>' + st.dist + ' кл.</b>');
    } else {
      st.enemies.forEach(en => {
        if (en.stun > 0) { en.stun--; this.log('💫 <b>' + en.name + '</b> оглушён.'); return; }
        const dmg = Math.max(1, en.dmg - p.armFlat);
        p.hp = Math.max(1, p.hp - dmg);
        window.BSIM_AUDIO.playSfx(en.sfx || 'bite');
        this.popBack('u-' + en.id, false, en.sound, 'float-sfx');
        this.popBack('u-player', true, '-' + dmg, 'float-dmg');
        this.log(en.sound + ' <b>' + en.name + '</b> бьёт: <b>-' + dmg + '</b> (броня -' + p.armFlat + ').');
      });
    }
    st.round++;
    this.renderBattle();
  }
};

window.BSIM_UI.init();
</script>
