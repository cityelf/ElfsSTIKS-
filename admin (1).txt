<script>
  function log(msg) {
    const b = document.getElementById('admin-log');
    b.innerHTML = '> ' + msg + '<br>' + b.innerHTML;
  }

  function getDB() {
    try {
      return JSON.parse(localStorage.getItem('stiks_prod_db')) || { families: {}, session: null };
    } catch (e) {
      return { families: {}, session: null };
    }
  }

  function saveDB(db) {
    localStorage.setItem('stiks_prod_db', JSON.stringify(db));
    if (window.STIKS) window.STIKS.db = db;
    renderAdmin();
  }

  function updatePatchStatus() {
    const el = document.getElementById('patch-status');
    const raw = localStorage.getItem('stiks_active_patch');
    if (raw) {
      try {
        const p = JSON.parse(raw);
        el.innerHTML = `Активен патч: <b style="color:#f0d78c;">${p.version || 'Custom'}</b> — ${p.desc || 'Без описания'}`;
      } catch (e) {
        el.innerText = 'Активен пользовательский патч.';
      }
    } else {
      el.innerHTML = 'Активен патч: <b style="color:#7fa88b;">Базовое ядро (v1.0)</b>';
    }
  }

  function pullServerPatch() {
    log('Запрос patch.js с сервера...');
    const old = document.getElementById('remote-patch-script');
    if (old) old.remove();
    const s = document.createElement('script');
    s.id = 'remote-patch-script';
    s.src = 'patch.js?t=' + Date.now();
    s.onload = function() {
      if (window.STIKS_REMOTE_PATCH && window.STIKS_PATCHER) {
        localStorage.setItem('stiks_active_patch', JSON.stringify(window.STIKS_REMOTE_PATCH));
        window.STIKS_PATCHER.apply(window.STIKS_REMOTE_PATCH);
        updatePatchStatus();
        log('✅ Патч с сервера успешно загружен и применён!');
      } else {
        log('⚠️ Файл patch.js загружен, но объект STIKS_REMOTE_PATCH не найден.');
      }
    };
    s.onerror = function() {
      log('❌ Ошибка загрузки patch.js (проверьте, залит ли файл в репозиторий).');
    };
    document.body.appendChild(s);
  }

  function applyManualPatch() {
    const val = document.getElementById('manual-patch').value.trim();
    if (!val) return;
    try {
      const clean = val.replace(/^window\.STIKS_REMOTE_PATCH\s*=\s*/, '').replace(/;$/, '');
      const obj = JSON.parse(clean);
      localStorage.setItem('stiks_active_patch', JSON.stringify(obj));
      if (window.STIKS_PATCHER) window.STIKS_PATCHER.apply(obj);
      updatePatchStatus();
      log('✅ Ручной патч сохранён и применён!');
      document.getElementById('manual-patch').value = '';
    } catch (e) {
      log('❌ Ошибка синтаксиса JSON в патче: ' + e.message);
    }
  }

  function clearLocalPatch() {
    if (confirm('Удалить локальный кэш патча и вернуться к базовому ядру?')) {
      localStorage.removeItem('stiks_active_patch');
      location.reload();
    }
  }

  function renderAdmin() {
    updatePatchStatus();
    const db = getDB();
    const rInfo = document.getElementById('raid-info');
    const rCtrl = document.getElementById('raid-controls');
    const rAct = document.getElementById('raid-actions');
    const lCard = document.getElementById('loot-card');

    if (db.session) {
      const s = db.session;
      rInfo.innerHTML = `Ход: <b>${s.turn}</b> | Игроков: <b>${s.players.length}</b> | Заражённых на карте: <b>${s.zombies.length}</b>`;
      rCtrl.style.display = 'flex';
      rAct.style.display = 'flex';
      lCard.style.display = 'block';
      document.getElementById('raid-diff').value = s.diffKey || 'start';

      const pSel = document.getElementById('loot-player');
      pSel.innerHTML = s.players.map((p, i) => `<option value="${i}">${p.family} ${p.name}</option>`).join('');

      const iSel = document.getElementById('loot-item');
      if (iSel.options.length === 0 && window.STIKS_DB) {
        const allItems = [];
        [1, 2, 3, 4].forEach(cls => {
          const block = window.STIKS_DB.ITEMS[cls];
          if (!block) return;
          ['weapon', 'armor', 'supply'].forEach(cat => {
            ['std', 'weak', 'troll', 'luck'].forEach(sub => {
              (block[cat][sub] || []).forEach(it => {
                allItems.push({ ...it, cat, cls });
              });
            });
          });
        });
        iSel.innerHTML = allItems.map((it, idx) => `<option value='${JSON.stringify(it)}'>[Кл.${it.cls}] ${it.n}</option>`).join('');
      }
    } else {
      rInfo.innerText = 'Активный рейд не запущен.';
      rCtrl.style.display = 'none';
      rAct.style.display = 'none';
      lCard.style.display = 'none';
    }

    const cList = document.getElementById('clans-list');
    const fams = Object.keys(db.families || {});
    if (fams.length === 0) {
      cList.innerHTML = '<div style="font-size:11px;color:#687a6e;">Кланы ещё не созданы.</div>';
    } else {
      let html = '';
      fams.forEach(fam => {
        html += `<div style="font-size:11px;color:#9ec4a9;margin:6px 0 4px 0;font-weight:bold;">КЛАН: ${fam}</div>`;
        db.families[fam].forEach((h, idx) => {
          html += `<div class="hero-item">
            <div><b>${h.name}</b> [Ур.${h.lvl}] (${h.status}) ${h.isInfected ? '☣️' : ''}</div>
            <div style="display:flex;gap:3px;">
              <button class="btn" style="padding:3px 6px;font-size:10px;" onclick="unlockHero('${fam}',${idx})">🔓</button>
              <button class="btn btn-primary" style="padding:3px 6px;font-size:10px;" onclick="lvlUpHero('${fam}',${idx})">+Ур</button>
              <button class="btn" style="padding:3px 6px;font-size:10px;" onclick="toggleInf('${fam}',${idx})">☣️</button>
              <button class="btn btn-danger" style="padding:3px 6px;font-size:10px;" onclick="delHero('${fam}',${idx})">Х</button>
            </div>
          </div>`;
        });
      });
      cList.innerHTML = html;
    }
  }

  function changeDiff() {
    const db = getDB();
    if (!db.session) return;
    const k = document.getElementById('raid-diff').value;
    db.session.diffKey = k;
    db.session.diffMod = window.STIKS_DB.DIFFS[k].mod;
    saveDB(db);
    log('Сложность рейда изменена на: ' + k);
  }

  function healSquad() {
    const db = getDB();
    if (!db.session) return;
    db.session.players.forEach(p => {
      p.hp = p.maxHp;
      p.food = 100;
      p.water = 100;
      p.stamina = 100;
      p.spore = 100;
      p.stinkTurns = 0;
    });
    saveDB(db);
    log('Весь отряд полностью исцелён и накормлен.');
  }

  function abortRaid() {
    if (!confirm('Завершить текущую сессию и разблокировать всех бойцов?')) return;
    const db = getDB();
    Object.values(db.families).forEach(arr => arr.forEach(h => { if (h.status === 'in_raid') h.status = 'ready'; }));
    db.session = null;
    saveDB(db);
    log('Активный рейд принудительно сброшен.');
  }

  function giveItem() {
    const db = getDB();
    if (!db.session) return;
    const pIdx = parseInt(document.getElementById('loot-player').value, 10);
    const it = JSON.parse(document.getElementById('loot-item').value);
    db.session.players[pIdx].inv.push(it);
    saveDB(db);
    log(`Выдан предмет [${it.n}] игроку ${db.session.players[pIdx].name}`);
  }

  function unlockHero(fam, idx) {
    const db = getDB();
    db.families[fam][idx].status = 'ready';
    saveDB(db);
    log('Статус бойца сброшен на ready.');
  }

  function lvlUpHero(fam, idx) {
    const db = getDB();
    const h = db.families[fam][idx];
    h.lvl += 1;
    h.maxHp += 5;
    h.hp = h.maxHp;
    saveDB(db);
    log(`Уровень бойца ${h.name} повышен до ${h.lvl}.`);
  }

  function toggleInf(fam, idx) {
    const db = getDB();
    const h = db.families[fam][idx];
    h.isInfected = !h.isInfected;
    saveDB(db);
    log(`Статус заражения ${h.name}: ${h.isInfected}`);
  }

  function delHero(fam, idx) {
    if (!confirm('Удалить бойца навсегда?')) return;
    const db = getDB();
    db.families[fam].splice(idx, 1);
    saveDB(db);
    log('Боец удалён.');
  }

  function testHive() {
    if (window.STIKS && window.STIKS.sayAs) {
      window.STIKS.sayAs('hive', 'Система Улья на связи. Проверка канала синхронизации.');
    }
  }

  function testAll() {
    if (window.STIKS && window.STIKS.testAllVoices) {
      window.STIKS.testAllVoices();
    }
  }

  function scanVoices() {
    const vs = window.speechSynthesis.getVoices().filter(v => v.lang.toLowerCase().includes('ru'));
    log('Найдено русских голосов: ' + vs.length);
    vs.forEach((v, i) => log(`[${i}] ${v.name}`));
  }

  function exportDB() {
    prompt('Скопируйте JSON базы данных:', localStorage.getItem('stiks_prod_db') || '{}');
  }

  function importDB() {
    const raw = prompt('Вставьте JSON базы данных:');
    if (!raw) return;
    try {
      const parsed = JSON.parse(raw);
      saveDB(parsed);
      log('База данных успешно импортирована.');
    } catch (e) {
      alert('Ошибка формата JSON!');
    }
  }

  function wipeDB() {
    if (confirm('ВНИМАНИЕ! Это удалит ВСЕ кланы, героев и прогресс. Продолжить?')) {
      localStorage.removeItem('stiks_prod_db');
      renderAdmin();
      log('База данных полностью очищена.');
    }
  }

  window.onload = renderAdmin;
</script>
